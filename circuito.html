Aquí se encuentra la configuración técnica del circuito de SAFEZONE: pines, programación y funcionamiento del sistema automatizado.

💻 Programación del circuito

SAFEZONE.ino
// ====================================================== // PROYECTO FINAL TICS - IRTRA PETAPA // SISTEMA DE SEGURIDAD AUTOMATIZADO // PARA ZONAS DE RESTRICCIÓN EN JUEGOS MECÁNICOS // // Madelin García, Teresa García, Mafer Cabrera - 11B // ====================================================== // ====================================================== // DEFINICIÓN DE PINES // ====================================================== #define LEDr 3 #define LEDa 5 #define LEDv 6 #define Trig A2 #define Echo A1 #define PIR A0 #define SDA A4 #define SCL A5 #define Button 7 #define Buzzer 12 // ====================================================== // LIBRERÍAS // ====================================================== #include <Wire.h> #include <LiquidCrystal_I2C.h> LiquidCrystal_I2C lcd(0x20, 16, 2); // ====================================================== // VARIABLES // ====================================================== long Tiempo; float Distancia; int Movimiento; int EstadoBoton; bool Alarma = false; int DistanciaSegura = 100; // ====================================================== // SETUP // ====================================================== void setup() { pinMode(LEDr, OUTPUT); pinMode(LEDa, OUTPUT); pinMode(LEDv, OUTPUT); pinMode(Trig, OUTPUT); pinMode(Echo, INPUT); pinMode(PIR, INPUT); pinMode(Button, INPUT); pinMode(Buzzer, OUTPUT); lcd.init(); lcd.backlight(); lcd.clear(); lcd.setCursor(0, 0); lcd.print("Sistema Activo"); lcd.setCursor(0, 1); lcd.print("Zona Segura"); digitalWrite(LEDv, HIGH); digitalWrite(LEDa, LOW); digitalWrite(LEDr, LOW); noTone(Buzzer); delay(2000); } // ====================================================== // FUNCIÓN PARA MEDIR DISTANCIA // ====================================================== void medirDistancia() { digitalWrite(Trig, LOW); delayMicroseconds(2); digitalWrite(Trig, HIGH); delayMicroseconds(10); digitalWrite(Trig, LOW); Tiempo = pulseIn(Echo, HIGH, 30000); if (Tiempo == 0) { Distancia = 999; } else { Distancia = Tiempo * 0.0343 / 2; } } // ====================================================== // LOOP // ====================================================== void loop() { // ==================================================== // MEDIR SENSORES // ==================================================== medirDistancia(); Movimiento = digitalRead(PIR); EstadoBoton = digitalRead(Button); // ==================================================== // SI NO ESTÁ EN ALARMA // ==================================================== if (Alarma == false) { // -------------------------------------------------- // SI DETECTA PELIGRO // -------------------------------------------------- if ( Distancia <= DistanciaSegura || Movimiento == HIGH ) { Alarma = true; digitalWrite(LEDv, LOW); lcd.clear(); lcd.setCursor(0, 0); lcd.print("PELIGRO!"); lcd.setCursor(0, 1); lcd.print("Zona 1 ALERTA"); delay(300); } // -------------------------------------------------- // SI TODO ESTÁ SEGURO // -------------------------------------------------- else { digitalWrite(LEDv, HIGH); digitalWrite(LEDa, LOW); digitalWrite(LEDr, LOW); noTone(Buzzer); lcd.setCursor(0, 0); lcd.print("Sistema Activo "); lcd.setCursor(0, 1); lcd.print("Zona Segura "); } } // ==================================================== // SI ESTÁ EN ALARMA // ==================================================== if (Alarma == true) { // -------------------------------------------------- // ACTIVAR ALARMA // -------------------------------------------------- digitalWrite(LEDv, LOW); digitalWrite(LEDa, LOW); digitalWrite(LEDr, HIGH); tone(Buzzer, 1000); lcd.setCursor(0, 0); lcd.print("PELIGRO! "); lcd.setCursor(0, 1); lcd.print("Presione RESET "); // -------------------------------------------------- // REVISAR BOTÓN // -------------------------------------------------- if (digitalRead(Button) == HIGH) { // Apagamos temporalmente la alarma noTone(Buzzer); digitalWrite(LEDr, LOW); lcd.clear(); lcd.setCursor(0, 0); lcd.print("VERIFICANDO..."); delay(500); // ------------------------------------------------ // VOLVER A MEDIR TODO // ------------------------------------------------ medirDistancia(); Movimiento = digitalRead(PIR); // ------------------------------------------------ // COMPROBAR SI ESTÁ SEGURO // ------------------------------------------------ if ( Movimiento == LOW && Distancia > DistanciaSegura ) { // ============================================== // ZONA SEGURA // ============================================== Alarma = false; noTone(Buzzer); digitalWrite(LEDr, LOW); digitalWrite(LEDa, LOW); digitalWrite(LEDv, HIGH); lcd.clear(); lcd.setCursor(0, 0); lcd.print("Sistema Activo"); lcd.setCursor(0, 1); lcd.print("Zona Segura"); delay(1500); } else { // ============================================== // TODAVÍA HAY PELIGRO // ============================================== Alarma = true; digitalWrite(LEDv, LOW); digitalWrite(LEDa, LOW); digitalWrite(LEDr, HIGH); tone(Buzzer, 1000); lcd.clear(); lcd.setCursor(0, 0); lcd.print("NO REINICIAR"); lcd.setCursor(0, 1); lcd.print("Zona ocupada!"); delay(2000); } } // -------------------------------------------------- // PARPADEO DEL LED ROJO // -------------------------------------------------- digitalWrite(LEDr, LOW); delay(250); digitalWrite(LEDr, HIGH); delay(250); } }

🧠 ¿Qué representa?

SAFEZONE integra sensores de distancia y movimiento con señales visuales, sonoras y una pantalla LCD para comunicar el estado del sistema al operador.

El sistema permanece en estado seguro mientras no se detecte movimiento ni una persona dentro de la distancia de seguridad. Cuando se detecta una situación de riesgo, se activa la alarma visual y sonora.

El Push Button permite comprobar nuevamente las condiciones de la zona antes de desactivar la alarma.
